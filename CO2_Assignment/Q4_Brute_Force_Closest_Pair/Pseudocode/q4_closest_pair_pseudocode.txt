ALGORITHM BruteForceClosestPair(P[0..n-1])
  min_dist <- INFINITY; best_pair <- NULL
  pairs_examined <- 0; dist_calcs <- 0
  FOR i <- 0 TO n - 2 DO
    FOR j <- i + 1 TO n - 1 DO
      pairs_examined <- pairs_examined + 1
      d <- SQRT((P[i].x - P[j].x)^2 + (P[i].y - P[j].y)^2)
      dist_calcs <- dist_calcs + 1
      IF d < min_dist THEN
        min_dist <- d; best_pair <- (P[i], P[j])
  RETURN best_pair, min_dist, pairs_examined, dist_calcs
